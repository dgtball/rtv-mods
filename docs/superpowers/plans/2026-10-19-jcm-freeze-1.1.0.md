# Freeze for JCM 1.1.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Добавить в аддон `jcm-freeze` тумблер «Freeze BTR & Heli» (заморозка BTR и вертолёта-ивента со заглушением звука двигателя BTR) и настраиваемый хоткей переключения Freeze NPC через MCM.

**Architecture:** Аддон остаётся набором replace-хуков modloader'а, зарегистрированных один раз; в колбэках решают булевы флаги. Новый модуль `FreezeMCM.gd` — тонкий слой над MCM (регистрация Keycode + чтение значения); `Main.gd` слушает `_input` и дёргает JCM `set_enabled`, поэтому тумблер и мир синхронны. BTR дополнительно глушит свои луп-плееры через `_lib._caller`, вертолёт замирает обычным пропуском `_physics_process`.

**Tech Stack:** GDScript (Godot 4.6.3), Metro Mod Loader `hook`/`skip_super`/`_caller`, JCM addon API 1.0 (`register_toggle`/`is_enabled`/`set_enabled`), MCM `MCM_Config` (`CreateKeycodeValue`/`RegisterMod`).

## Global Constraints

- Vерсия аддона — `loadouts/mod.txt` Востока Тулкита не трогать; версия живёт в `jcm-freeze/mod.txt`, после изменений = `1.1.0`.
- Отступы в `.gd` — только табы; файлы UTF-8; тексты в коде — английские; комментарии не добавлять без нужды (упаковщик их всё равно срезает).
- Headless-тестов аддона нет: вся логика — рантайм-хуки и API соседей. Верификация = parse-check + pack + in-game gate.
- Коммиты **не делать**, пока пользователь явно не разрешил (правило репозитория имеет приоритет над шаблоном шагов скилла).
- Имена replace-хуков обязаны быть литеральными строками в `.hook("...")` и продублированы в `[hooks]` mod.txt (контракт wrap-поверхности `modloader.gd:14510–14514`).
- Replace-колбэк вызывается `callv(args)`; арность должна совпадать с методом игры точно (иначе молчаливый no-op).
- Хуки занимают глобальные replace-слоты: `_npc_on`/`_vehicles_on` — единственное состояние, никаких операций над чужими узлами вне колбэков.
- Парсинг-стенд для модулей, ссылающихся на MCM (`MCM_Config`), требует заглушки в `.check/project/stubs/`.

- [ ] **Step 2: Поглощение и восстановление**

После `_on_heli_process` добавить:

```gdscript
func _silence_btr() -> void:
	var btr = _lib._caller
	if btr == null or not is_instance_valid(btr):
		return
	var key = btr.get_instance_id()
	if _bt_seen.has(key):
		return
	_bt_seen[key] = true
	for name in ["Idle", "Drive", "Road"]:
		var player = btr.get_node_or_null("Audio/" + name)
		if player == null:
			continue
		_bt_audio.append({"node": player, "volume_db": player.volume_db, "playing": player.playing})
		player.stop()


func _restore_btr() -> void:
	for rec in _bt_audio:
		var player = rec["node"]
		if not is_instance_valid(player):
			continue
		player.volume_db = rec["volume_db"]
		if rec["playing"]:
			player.play()
	_bt_audio.clear()
	_bt_seen.clear()
```

- [ ] **Step 2: Новый модуль `FreezeMCM.gd`**

Создать `jcm-freeze/jcm-freeze/FreezeMCM.gd`:

```gdscript
extends Node

const MOD_ID = "jcm-freeze"
const KEY_ID = "freezeNpcKey"

var key_freeze_npc: Key = KEY_KP_4
var _mcm_config = null


func _ready() -> void:
	if not Engine.has_meta("RTVModLib"):
		return
	var lib = Engine.get_meta("RTVModLib")
	if lib.get("_is_ready"):
		_register()
	else:
		lib.frameworks_ready.connect(_register)


func _register() -> void:
	var lib = Engine.get_meta("RTVModLib")
	if not lib.has_mod("doinkoink-mcm"):
		return

	_mcm_config = MCM_Config.new(MOD_ID, "Freeze for JCM",
			"Freeze NPCs and vehicles on demand.", _on_mcm_saved, self)

	_mcm_config.CreateKeycodeValue(KEY_ID, "Toggle Freeze NPC",
			"Turns Freeze NPC on or off without opening the menu.",
			key_freeze_npc, MCM_Config.MCM_Key_Types.KEY) \
		.setCategory("Hotkeys")

	_mcm_config.RegisterMod()


func _on_mcm_saved(config: ConfigFile) -> void:
	key_freeze_npc = _mcm_value(config, "Keycode", KEY_ID, key_freeze_npc)


func _mcm_value(config: ConfigFile, section: String, key: String, fallback):
	if config.has_section_key(section, key):
		var entry = config.get_value(section, key)
		if entry is Dictionary and entry.has("value"):
			return entry["value"]
	return fallback
```

```gdscript
func _input(event: InputEvent) -> void:
	if _jcm == null or _mcm == null:
		return
	if not (event is InputEventKey or event is InputEventMouseButton) or not event.pressed:
		return
	if not _hotkey_pressed(event, "freezeNpcKey", _mcm.key_freeze_npc):
		return
	if get_viewport().gui_get_focus_owner() is LineEdit:
		return
	var gd = _jcm.get("gameData")
	if gd == null or gd.menu or gd.settings:
		return
	_jcm.set_enabled(NPC_ID, not _jcm.is_enabled(NPC_ID))


func _hotkey_pressed(event: InputEvent, action: String, fallback_keycode: int) -> bool:
	if InputMap.has_action(action):
		return event.is_action_pressed(action)
	return event is InputEventKey and not event.echo and event.physical_keycode == fallback_keycode
```
