## Как читать этот план

Для инженера, который не знает ни игры, ни мода. Каждая задача — законченный
шаг с прогоном стенда и коммитом. Термины:

- **NVG-узел** — ванильный `Control` `NVG.gd` на `/root/Map/Core/UI/NVG`; его
  `Overlay.get_child(0).material` — тот самый `ShaderMaterial`, в который мод
  подменяет `.shader`, чтобы получить управляемый uniform `noise`.
- **Тепловизор** (`on`) — `set_thermal and _nvg_active()`, где `_nvg_active()` =
  `gameData.NVG`. Это НЕ то же самое, что включённый прибор: `set_thermal` —
  настройка мода, `gameData.NVG` — факт включения NVG в игре.
- **`node_added`** — сигнал `SceneTree`, приходит на каждый узел, вошедший в
  дерево; у мода на нём висит детект ботов/мин/NVG.
- **`_deferred_check`** — существующий метод, который по `instance_id` достаёт
  узел и классифицирует его (бот/мина/трейдер/NVG).

### Тесты и гейт (важно)

Правки Batch 3 — рантайм-кадровые (`SceneTree`/`RenderingServer`), чистой
логики не появляется. Поэтому **headless-тесты не добавляем** (решение
владельца, спека §«Тесты и проверка»). Гейт каждой задачи:

```powershell

py .check\layout_check.py
```

Зелёный `check.ps1`: парсинг 16 скриптов + строка `tests passed` в каждом
тест-файле, код выхода 0, и НИ ОДНОЙ строки `SCRIPT ERROR` / `Parse Error` /
`Invalid call` / `Failed to load script`. Зелёный `layout_check.py`:
`layout exit: 0`. Существующие тесты (`test_feed.gd`, `test_hooks.gd` и т. д.)
обязаны остаться зелёными — они и есть регрессионная сеть.

- [ ] **2.4 `_update_bot`.** Замени тело (629-697) на версию с `ppos`, кэшем
      фракции, свипом по числу детей и общим death-clock полем `recorded`:

```gdscript
func _update_bot(id: int, thermal_on: bool, ppos: Vector3) -> void:
	var e = _bots.get(id)
	if e == null:
		return
	var node = e.get("node")
	if node == null or not is_instance_valid(node):
		_bots.erase(id)
		return

	# Фракция приходит из AIData variant, который в момент регистрации может
	# быть null. Пересчитываем, пока variant не готов; как только резолвился —
	# кэш постоянен и _is_friendly больше не зовётся.
	if not e["friendly_resolved"]:
		e["friendly"] = _is_friendly(node)
		var v = node.get("variant")
		if v != null and v.get("faction") != null:
			e["friendly_resolved"] = true
	var friendly: bool = e["friendly"]

	# Death clock: лэтчим момент смерти один раз, чтобы фронт остывания не
	# сбрасывался на кадрах, когда тепловизор выключен (см. _off_frame_upkeep).
	var now := float(Time.get_ticks_msec()) / 1000.0
	if bool(node.get("dead")):
		if not e["recorded"]:
			e["recorded"] = true
			e["death"] = now
	else:
		e["recorded"] = false
	var secs := 0.0
	if e["recorded"]:
		secs = now - float(e["death"])

	# Дружественный Nomad получает ▼ над головой, но только внутри радиуса
	# свечения -- иначе маркер читался бы через всю карту.
	var mark := false
	if friendly and not e["recorded"]:
		mark = thermal_on and set_mark_nomads \
			and ppos.distance_to(node.global_position) <= maxf(set_far_distance, 1.0)
	_update_nomad_marker(node, mark, ppos)

	# Геометрия бота меняется, когда игра добавляет/убирает LOD и gear-меши.
	# Полный свип каждый кадр был расточителен; пере-собираем только когда
	# число детей реально изменилось -- это ловит любое добавление/удаление,
	# а стабильное тело (обычный случай) бесплатно.
	var cc := node.get_child_count()
	if int(e["child_count"]) != cc:
		e["meshes"] = _sweep_bot_meshes(node)
		e["child_count"] = cc

	# Per-bot body material off the shared shader: "cool" and "footY" are
	# per-bot uniforms. cool = secs / DEATH_COOLDOWN drives the cooling front
	# (VTKBot.gdshader) from just above the head down past the feet.
	var bmat: ShaderMaterial = e.get("mat")
	if bmat == null and _bot_shader != null:
		bmat = ShaderMaterial.new()
		bmat.shader = _bot_shader
		bmat.set_shader_parameter("warmKey", BOT_WARM_KEY)
		bmat.set_shader_parameter("midKey", BOT_MID_KEY)
		bmat.set_shader_parameter("coldKey", BOT_COLD_KEY)
		e["mat"] = bmat
	if bmat != null:
		var cool := 0.0 if not e["recorded"] else clampf(secs / DEATH_COOLDOWN, 0.0, 1.0)
		bmat.set_shader_parameter("cool", cool)
		bmat.set_shader_parameter("footY", node.global_position.y)

	# A body only glows inside the glow range and while it still has heat left;
	# past either bound the override is cleared and the real skin shows through.
	var dist := ppos.distance_to(node.global_position)
	var show := thermal_on and bmat != null and dist <= maxf(set_far_distance, 1.0) \
		and (not e["recorded"] or secs < DEATH_COOLDOWN)
	e["painted"] = show
	if show:
		_paint_on = true
	var paint = bmat if show else null
	e["paint"] = paint
	for m in e["meshes"]:
		if is_instance_valid(m):
			(m as GeometryInstance3D).material_override = paint
```

```gdscript
# Покой при выключенном тепловизоре. Регистрация (node_added) продолжает идти
# и при off, поэтому освобождённые записи всё равно надо чистить, иначе _mines
# рос бы без границы. Заодно лэтчим death clock: иначе труп, убитый при
# выключенном тепловизоре, при включении получил бы death=now и светился бы
# лишние DEATH_COOLDOWN секунд.
func _off_frame_upkeep() -> void:
	var now := float(Time.get_ticks_msec()) / 1000.0
	for id in _bots.keys():
		var e = _bots.get(id)
		if e == null:
			continue
		var node = e.get("node")
		if node == null or not is_instance_valid(node):
			_bots.erase(id)
			continue
		if bool(node.get("dead")):
			if not e["recorded"]:
				e["recorded"] = true
				e["death"] = now
		else:
			e["recorded"] = false
	for id in _mines.keys():
		var e = _mines.get(id)
		if e == null or not is_instance_valid(e.get("node")):
			_mines.erase(id)


# Снимает покраску на переходе on->off ровно один раз: material_override в
# null на все меши ботов и мин, маркеры Nomad скрыты. Дальше кадр при off
# ботов и мешей не обходит вовсе.
func _clear_paint() -> void:
	for id in _bots.keys():
		var e = _bots.get(id)
		if e == null:
			continue
		for m in e.get("meshes", []):
			if is_instance_valid(m):
				(m as GeometryInstance3D).material_override = null
		e["painted"] = false
		e["paint"] = null
		var node = e.get("node")
		if node != null and is_instance_valid(node):
			_update_nomad_marker(node, false, Vector3.ZERO)
	for id in _mines.keys():
		var e = _mines.get(id)
		if e == null:
			continue
		for m in e.get("meshes", []):
			if is_instance_valid(m):
				(m as GeometryInstance3D).material_override = null
```
