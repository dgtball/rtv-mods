# Batch 4 — UI-пересборки: единый refresh, дебаунс поиска, кэши, скелет клеток, снятие спавнера — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Убрать лишнюю работу при перерисовке UI `VTKUI.gd` — единый scoped `refresh`, дебаунс поиска, кэши раскладки и стилей, общий скелет клеток — и снять кнопку спавна предмета в обход сэта.

**Architecture:** Пять независимых правок в одном файле (`VTKUI.gd`) плюс снятие спавнера в `VTKUI.gd`/`VTKData.gd`/`VTKLoc.gd`. Правки упорядочены от фундамента к частностям: сначала единый `refresh(Scope)` (на него опирается дебаунс поиска), потом снятие спавнера (упрощает клетку), затем кэши и скелет. Каждая задача оставляет продукт собираемым и проверяемым стендами.

**Tech Stack:** Godot 4.6.3 (GDScript, табы), PowerShell-стенды `.check/check.ps1` (парсинг + headless-тесты) и `.check/layout_check.py` (раскладка по исходникам).

```gdscript
# Единый вход перерисовки: область задаётся битмаской RefreshScope. Шаг «±»
# не пересобирает экипировку, закрытие обвесов - каталог.
class RefreshScope:
	const EQUIP := 1
	const GEAR := 2
	const DONE := 4
	const OVERLAY := 8
	const LIST := 16
	const PREVIEW := 32
	const PAGES := EQUIP | GEAR | DONE | OVERLAY

func refresh(scope: int) -> void:
	if scope & RefreshScope.EQUIP:
		_refresh_page_equipment()
	if scope & RefreshScope.GEAR:
		_refresh_gear_catalog()
	if scope & RefreshScope.DONE:
		_refresh_page_done()
	if scope & RefreshScope.OVERLAY:
		_refresh_att_overlay()
	if scope & RefreshScope.LIST:
		_refresh_loadout_list()
	if scope & RefreshScope.PREVIEW:
		_refresh_preview()
```

```gdscript
# Один предмет снаряжения в инвентарь целиком (count обойм или amount
# копий): кнопка «В инвентарь» в клетке шага 2. Без игрового дерева молча
# 0, а не ошибка - так же ведёт себя apply_loadout, и headless-тест не
# должен падать.
static func spawn_gear(desc: Dictionary) -> int:
	var root = _root_node()
	if not root:
		return 0
	var interface = root.get_node_or_null(Paths.INTERFACE)
	if not interface:
		Log.log("[VTK] spawn_gear: no Interface node")
		return 0
	var loader = root.get_node_or_null("Loader")
	var ig = interface.get("inventoryGrid")
	if ig == null or not is_instance_valid(ig):
		Log.log("[VTK] spawn_gear: no inventoryGrid")
		return 0
	var placed := _insert_desc(interface, ig, desc)
	if loader and loader.has_method("SaveCharacter"):
		loader.SaveCharacter()
	if interface.has_method("UpdateStats"):
		interface.UpdateStats(true)
	if placed > 0:
		var res = load(str(desc.get("path", "")))
		var prefix := Loc.txt("Spawned to catalog: ") if _is_furniture(res) else Loc.txt("Spawned: ")
		_notify(loader, prefix + str(desc.get("name", "?"))
			+ " x" + str(placed), Color.GREEN)
	return placed
```

- [ ] **Step 1: Добавить `_art_texture_rect` и `_icon_stack`**

Перед `_make_item_cell` добавить:

```gdscript
# Иконка предмета: кроп прозрачных полей через Picker.item_art и режим
# сохранения пропорций. Общая для клеток шага 1/2 и чипа инвентаря.
func _art_texture_rect(art: Texture2D, min_size: Vector2) -> TextureRect:
	var tex := TextureRect.new()
	tex.texture = Picker.item_art(art)
	tex.stretch_mode = TextureRect.STRETCH_KEEP_ASPECT_CENTERED
	tex.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	tex.custom_minimum_size = min_size
	tex.size_flags_horizontal = Control.SIZE_SHRINK_CENTER
	tex.mouse_filter = Control.MOUSE_FILTER_IGNORE
	return tex

# Иконка над подписью, по центру клетки: CenterContainer на всю клетку, VBox,
# TextureRect и подпись. Возвращает VBox, чтобы клетка дописала свои ряды.
func _icon_stack(host: Control, art: Texture2D, icon_min: Vector2, name_text: String,
		name_size: int, name_color: Color, name_lines: int) -> VBoxContainer:
	var center := CenterContainer.new()
	center.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	center.size_flags_vertical = Control.SIZE_EXPAND_FILL
	center.mouse_filter = Control.MOUSE_FILTER_IGNORE
	host.add_child(center)
	center.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	var vbox := VBoxContainer.new()
	vbox.add_theme_constant_override("separation", 2)
	vbox.mouse_filter = Control.MOUSE_FILTER_IGNORE
	center.add_child(vbox)
	var tex := _art_texture_rect(art, icon_min)
	vbox.add_child(tex)
	var name_lbl := Label.new()
	name_lbl.text = name_text
	name_lbl.add_theme_font_size_override("font_size", name_size)
	name_lbl.add_theme_color_override("font_color", name_color)
	name_lbl.autowrap_mode = TextServer.AUTOWRAP_WORD
	name_lbl.max_lines_visible = name_lines
	name_lbl.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	name_lbl.mouse_filter = Control.MOUSE_FILTER_IGNORE
	vbox.add_child(name_lbl)
	return vbox
```
